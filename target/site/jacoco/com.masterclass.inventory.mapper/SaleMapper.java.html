package com.masterclass.inventory.mapper; import com.masterclass.inventory.dto.SaleDtos.*; import com.masterclass.inventory.entity.*; import org.mapstruct.*; import java.util.*;
@Mapper(componentModel="spring") public interface SaleMapper { @Mapping(target="details", expression="java(toDetailResponses(sale.getDetails()))") SaleResponse toResponse(Sale sale); default List<SaleDetailResponse> toDetailResponses(List<SaleDetail> details){ return details.stream().map(d -> new SaleDetailResponse(d.getProduct().getId(),d.getProduct().getSku(),d.getProduct().getName(),d.getQuantity(),d.getUnitPrice(),d.getSubtotal())).toList(); } }
